  async function loadUniversalAssistantContext(activeUser: ActiveUserContext) {
    if (!supabaseAdmin) throw new Error('School data service is unavailable.');
    const [schoolRead, userRead, notificationCountRead] = await Promise.all([
      supabaseAdmin.from('schools')
        .select('id,school_name,school_code,address,phone_number,email')
        .eq('id', activeUser.schoolId).maybeSingle(),
      supabaseAdmin.from('users')
        .select('id,full_name,username')
        .eq('id', activeUser.userId).maybeSingle(),
      supabaseAdmin.from('edunixo_user_notifications')
        .select('id', { count: 'exact', head: true })
        .eq('school_id', activeUser.schoolId)
        .eq('recipient_user_id', activeUser.userId)
        .eq('is_read', false)
    ]);
    if (schoolRead.error) throw schoolRead.error;
    if (userRead.error) throw userRead.error;
    const school: any = schoolRead.data || {};
    const user: any = userRead.data || {};
    return {
      school: {
        id: activeUser.schoolId,
        name: safePublicText(school.school_name, 180) || 'Classtago School',
        code: safePublicText(school.school_code, 80),
        address: safePublicText(school.address, 300),
        phone: safePublicText(school.phone_number, 80),
        email: safePublicText(school.email, 180)
      },
      user: {
        id: activeUser.userId,
        name: safePublicText(user.full_name || user.username, 180) || 'School user',
        role: activeUser.role
      },
      unreadNotifications: Number(notificationCountRead.count || 0)
    };
  }

  function universalPersonaInstruction(persona: string, role: string) {

      ? 'Sound like a supportive school mentor: natural, age-appropriate and concise.'
      : 'Sound like a warm, capable school colleague speaking one-to-one.';
  }

        const campaign=await supabaseAdmin.from('school_admission_campaigns').select('id,title,session_name,status,start_date,end_date').eq('school_id',activeUser.schoolId).eq('status','active').order('created_at',{ascending:false}).limit(1).maybeSingle();
        return {status:'ok',school:school.school,admissionCampaign:campaign.error?null:campaign.data||null,note:'Maria must answer only from verified school data; missing fees/timings/policies must not be invented.'};
      }

      return {status:'open_module',message:'This feature is available through its existing Classtago module.',route:mariaFeatureRoute(activeUser,feature)};
    } catch (error:any) {
      return {status:'unavailable',feature,message:safePublicText(error?.message,300)||'This live ERP summary is not available right now.',route:mariaFeatureRoute(activeUser,feature)};
    }
  }

  async function prepareHeadmasterDialTool(activeUser: ActiveUserContext, requestedName: string) {
    if (!supabaseAdmin) throw new Error('Staff directory is unavailable.');
    if (activeUser.role !== 'headmaster') return { result: { status: 'forbidden', message: 'Only an authorised Headmaster can prepare a staff dial handoff.' } };
    const target = normalizeAssistantContactName(safePublicText(requestedName, 100));
    if (!target) return { result: { status: 'needs_name', message: 'Please provide the staff member name.' } };
    const staffRead = await supabaseAdmin.from('teachers')
      .select('id,user_id,full_name,name,designation,mobile_number,is_active')
      .eq('school_id', activeUser.schoolId).eq('is_active', true).limit(1200);
    if (staffRead.error) throw staffRead.error;
    const targetTokens = target.split(' ').filter(Boolean);
    const scored = (staffRead.data || []).map((row: any) => {
      const displayName = String(row.full_name || row.name || '').trim();
      const normalized = normalizeAssistantContactName(displayName);
      const candidateTokens = new Set(normalized.split(' ').filter(Boolean));
      const overlap = targetTokens.filter(token => candidateTokens.has(token)).length;
      let score = 0;
      if (normalized === target) score = 100;
      else if (normalized.startsWith(target) || target.startsWith(normalized)) score = 90;
      else if (normalized.includes(target) || target.includes(normalized)) score = 82;
      else if (targetTokens.length && overlap === targetTokens.length) score = 76;
      else if (overlap > 0) score = 45 + overlap * 8;
      return { row, displayName, score };
    }).filter((item: any) => item.displayName && item.score >= 60)
      .sort((a: any,b: any) => b.score - a.score || a.displayName.localeCompare(b.displayName));
    if (!scored.length) return { result: { status: 'not_found', message: `No active staff match was found for ${requestedName}.` } };
    const bestScore = scored[0].score;
    const close = scored.filter((item: any) => item.score >= bestScore - 4).slice(0, 4);
    if (close.length > 1) return { result: { status: 'ambiguous', candidates: close.map((item: any) => item.displayName) } };
    const selected = close[0];
    let phone = safePublicText(selected.row.mobile_number, 60);
    if (!phone && selected.row.user_id) {
      const userPhone = await supabaseAdmin.from('users').select('phone_number').eq('id', selected.row.user_id).maybeSingle();
      if (!userPhone.error) phone = safePublicText(userPhone.data?.phone_number, 60);
    }
    const dialable = String(phone || '').replace(/[^+0-9]/g, '');
    if (!dialable || dialable.replace(/\D/g, '').length < 7) {
      return { result: { status: 'missing_number', contactName: selected.displayName, message: 'No active mobile number is stored for this staff member.' } };
    }
    return {
      result: { status: 'ready', contactName: selected.displayName, action: 'open_device_dialer' },
      clientAction: { type: 'dial', label: `Call ${selected.displayName}`, uri: `tel:${dialable}`, contactName: selected.displayName }
    };
  }

  async function executeUniversalLiveTool(activeUser: ActiveUserContext, name: string, args: any) {
    if (!supabaseAdmin) throw new Error('Classtago tool gateway is unavailable.');
    if (name === 'get_current_school_context') {
      const context = await loadUniversalAssistantContext(activeUser);
      return { result: { assistantName:'Maria', school: context.school, user: context.user, unreadNotifications: context.unreadNotifications } };
    }
    if (name === 'get_latest_notifications') {
      const limit = Math.min(10, Math.max(1, Number(args?.limit || 5)));
      const read = await supabaseAdmin.from('edunixo_user_notifications')
        .select('title,body,notification_type,is_read,created_at')
        .eq('school_id', activeUser.schoolId).eq('recipient_user_id', activeUser.userId)
        .order('created_at', { ascending: false }).limit(limit);
      if (read.error) throw read.error;
      return { result: { notifications: (read.data || []).map((row: any) => ({ title:safePublicText(row.title,220), body:safePublicText(row.body,700), type:safePublicText(row.notification_type,60), isRead:Boolean(row.is_read), createdAt:row.created_at||null })) } };
    }
    if (name === 'get_feature_summary') {
      return { result: await mariaFeatureSummary(activeUser, String(args?.feature || ''), args) };
    }
    if (name === 'open_edunixo_feature') {
      const feature=String(args?.feature||'').trim().toLowerCase().replace(/[\s-]+/g,'_');
      const route=mariaFeatureRoute(activeUser,feature);
      if(!route) return {result:{status:'forbidden',message:'That Classtago feature is not available to this signed-in role.'}};
      return {
        result:{status:'ready',feature,moduleId:route.moduleId,featureId:route.featureId||null,message:'The authorised Classtago module is ready to open.'},
        clientAction:{type:'navigate',label:'Open in Classtago',moduleId:route.moduleId,featureId:route.featureId||undefined,feature}
      };
    }
    if (name === 'get_my_teaching_assignments') {
      if (!['teacher','class_teacher','headmaster'].includes(activeUser.role)) return { result: { status:'forbidden',message:'Teaching assignments are not available to this role.' } };
      try {
        const comm:any=await loadTeacherCommunicationScopes(activeUser);
        return {result:{academicYear:String(comm.academicYear||'Current Academic Year'),assignments:(comm.scopes||[]).filter((scope:any)=>scope.scopeType==='subject').slice(0,80).map((scope:any)=>({assignmentId:String(scope.assignmentId||''),className:String(scope.className||''),division:String(scope.division||''),subjectName:String(scope.subjectName||''),isClassTeacher:Boolean(scope.isClassTeacherScope)}))}};
      } catch(error:any) { return {result:{status:'unavailable',message:safePublicText(error?.message,240)||'Teaching assignments could not be loaded.'}}; }
    }
    if (name === 'prepare_staff_dial') return prepareHeadmasterDialTool(activeUser, String(args?.name || ''));
    return { result: { status:'unsupported',message:`Maria tool ${safePublicText(name,80)} is not enabled.` } };
  }

  function sanitizeRealtimeClientPayload(payload: any) {
    const input = payload?.realtimeInput;
    if (!input || typeof input !== 'object') return null;
    const realtimeInput: any = {};
    if (input.audio && typeof input.audio === 'object') {
      const data = String(input.audio.data || '');
      const mimeType = String(input.audio.mimeType || '');
      if (!data || data.length > 350_000 || mimeType !== 'audio/pcm;rate=16000') return null;
      realtimeInput.audio = { data, mimeType };
    }
    if (typeof input.text === 'string') {
      const text = safePublicText(input.text, 4000);
      if (text) realtimeInput.text = text;
    }
    if (input.audioStreamEnd === true) realtimeInput.audioStreamEnd = true;
    return Object.keys(realtimeInput).length ? { realtimeInput } : null;
  }
